  set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<IF:$<AND:$<C_COMPILER_ID:MSVC>,$<CXX_COMPILER_ID:MSVC>>,$<$<CONFIG:Debug,RelWithDebInfo>:EditAndContinue>,$<$<CONFIG:Debug,RelWithDebInfo>:ProgramDatabase>>")

project("UnleashedRecomp-ALL")

cmake_policy(SET CMP0069 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0069 NEW)
if (CMAKE_BUILD_TYPE STREQUAL "Release" AND NOT UNLEASHED_RECOMP_SWITCH)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT _IPO_OK OUTPUT _IPO_ERROR)
    if (_IPO_OK)
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
        message(STATUS "LTO enabled for Release build")
    else()
        message(STATUS "LTO not available: ${_IPO_ERROR}")
    endif()
endif()

# Link-time optimisation for the Switch build. It was left off because linking the recompiled
# game with LTO needs a lot of RAM on the build machine; nfsmw-nx (same toolchain, 144 MB of
# recompiled C++) links it with 2 LTO jobs and balanced partitions. Its benefit multiplies with
# tools/switch-direct-calls.py, which lets GCC inline across recompiled functions at all.
option(UNLEASHED_RECOMP_SWITCH_LTO "Build the Switch NRO with link-time optimisation." OFF)
set(UNLEASHED_RECOMP_SWITCH_LTO_JOBS "2" CACHE STRING "Parallel LTRANS jobs for the Switch LTO link.")

if (UNLEASHED_RECOMP_SWITCH AND UNLEASHED_RECOMP_SWITCH_LTO)
    message(STATUS "Switch LTO enabled (${UNLEASHED_RECOMP_SWITCH_LTO_JOBS} jobs)")
    add_compile_options(-flto -fno-fat-lto-objects)
    add_link_options(-flto=${UNLEASHED_RECOMP_SWITCH_LTO_JOBS} -flto-partition=balanced)
    add_compile_definitions(UNLEASHED_RECOMP_SWITCH_LTO)
endif()

# Round 9 (tools/build-switch.sh SWITCH_O2=1, SWITCH_IPA_PTA=1): -O2 instead of CMake's release -O3 for all the
# code (with PGO, GCC turns on the -O3 loop and inlining work for the hot code by itself), and interprocedural
# points-to analysis, which runs on the whole program at the LTO link (so it goes to the link as well).
option(UNLEASHED_RECOMP_SWITCH_O2 "Compile all Switch code with -O2 instead of -O3." OFF)
option(UNLEASHED_RECOMP_SWITCH_IPA_PTA "Compile and link the Switch build with -fipa-pta." OFF)
if (UNLEASHED_RECOMP_SWITCH AND UNLEASHED_RECOMP_SWITCH_O2)
    message(STATUS "Switch: -O2")
    string(REPLACE "-O3" "-O2" CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE}")
    string(REPLACE "-O3" "-O2" CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE}")
    string(REPLACE "-O3" "-O2" CMAKE_C_FLAGS_RELWITHDEBINFO "${CMAKE_C_FLAGS_RELWITHDEBINFO}")
    string(REPLACE "-O3" "-O2" CMAKE_CXX_FLAGS_RELWITHDEBINFO "${CMAKE_CXX_FLAGS_RELWITHDEBINFO}")
endif()
if (UNLEASHED_RECOMP_SWITCH AND UNLEASHED_RECOMP_SWITCH_IPA_PTA)
    message(STATUS "Switch: -fipa-pta")
    add_compile_options(-fipa-pta)
    add_link_options(-fipa-pta)
endif()

# tools/build-switch.sh SWITCH_CLASSIC_CODEGEN=1: ppc/ generated by XenonRecomp in its pre-round-7 mode
# (XENON_RECOMP_CLASSIC), whose code expects guest memory accesses to be volatile (ppc_context.h); the app
# reads the same macros. For A/B measurements of the round-7 code generation.
option(UNLEASHED_RECOMP_SWITCH_CLASSIC_CODEGEN "ppc/ was generated with the pre-round-7 code generation." OFF)
if (UNLEASHED_RECOMP_SWITCH AND UNLEASHED_RECOMP_SWITCH_CLASSIC_CODEGEN)
    message(STATUS "Switch: classic guest code generation")
    add_compile_definitions(PPC_VOLATILE_GUEST_MEMORY UNLEASHED_RECOMP_CLASSIC_CODEGEN)
endif()
